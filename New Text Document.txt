    <script>
        // -- APP DATA (Real-time Simulation) --
        let products = [
            { id: 1, name: "برنامج برو", price: 1500, color: "bg-blue-600", stock: 10 },
            { id: 2, name: "ماكينة كاشير", price: 4500, color: "bg-black", stock: 5 },
            { id: 3, name: "طابعة إيصالات", price: 2200, color: "bg-slate-700", stock: 8 },
            { id: 4, name: "ورق حراري (كرتونة)", price: 350, color: "bg-indigo-500", stock: 20 },
            { id: 5, name: "صيانة دورية", price: 500, color: "bg-gray-800", stock: 99 },
            { id: 6, name: "تفعيل سحابي", price: 1200, color: "bg-blue-900", stock: 100 }
        ];

        let cart = [];
        let salesHistory = [];
        let totalRevenue = 0;

        // -- NAVIGATION --
        function switchTab(tab) {
            document.getElementById('pos-screen').classList.toggle('hidden', tab !== 'pos');
            document.getElementById('admin-screen').classList.toggle('hidden', tab !== 'admin');
            
            document.getElementById('tab-pos').className = (tab === 'pos') ? 'pb-1 transition-all tab-active' : 'pb-1 transition-all text-gray-500 hover:text-black';
            document.getElementById('tab-admin').className = (tab === 'admin') ? 'pb-1 transition-all tab-active' : 'pb-1 transition-all text-gray-500 hover:text-black';
            
            if(tab === 'admin') updateAdminDashboard();
        }

        // -- POS LOGIC --
        function renderPOS() {
            const grid = document.getElementById('pos-grid');
            grid.innerHTML = products.map(p => `
                <div onclick="addToCart(${p.id})" class="group cursor-pointer">
                    <div class="${p.color} h-32 rounded-3xl shadow-lg flex flex-col items-center justify-center p-4 transition-all group-active:scale-95 group-hover:shadow-2xl">
                        <span class="text-white font-bold text-center leading-tight mb-2">${p.name}</span>
                        <div class="bg-white/20 text-white text-[10px] px-2 py-1 rounded-full font-bold">${p.price} ج.م</div>
                    </div>
                </div>
            `).join('');
        }

        function addToCart(id) {
            const p = products.find(x => x.id === id);
            const inCart = cart.find(x => x.id === id);
            if(inCart) inCart.qty++;
            else cart.push({...p, qty: 1});
            updateCartUI();
        }

        function updateCartUI() {
            const container = document.getElementById('cart-items');
            const totalEl = document.querySelector('.text-2xl.font-black span:last-child');
            const taxEl = document.getElementById('tax-amount');
            const countEl = document.getElementById('cart-count');

            if(cart.length === 0) {
                container.innerHTML = `<div class="text-center text-gray-400 mt-20">السلة فارغة.. ابدأ بإضافة منتجات</div>`;
                totalEl.innerText = "0.00 ج.م";
                taxEl.innerText = "0.00 ج.م";
                countEl.innerText = "0";
                return;
            }

            container.innerHTML = cart.map(item => `
                <div class="flex items-center justify-between bg-gray-50 p-4 rounded-2xl border border-gray-100">
                    <div>
                        <h4 class="font-bold text-sm">${item.name}</h4>
                        <p class="text-[10px] text-gray-500">${item.qty} وحدة × ${item.price} ج.م</p>
                    </div>
                    <div class="flex items-center gap-3">
                        <span class="font-bold text-sm text-blue-600">${item.price * item.qty}</span>
                        <button onclick="removeFromCart(${item.id})" class="w-6 h-6 bg-red-50 text-red-500 rounded-full text-xs flex items-center justify-center font-bold">×</button>
                    </div>
                </div>
            `).join('');

            let subtotal = cart.reduce((s, i) => s + (i.price * i.qty), 0);
            let tax = subtotal * 0.05;
            let total = subtotal + tax;

            totalEl.innerText = total.toLocaleString() + " ج.م";
            taxEl.innerText = tax.toLocaleString() + " ج.م";
            countEl.innerText = cart.reduce((s, i) => s + i.qty, 0);
        }

        function removeFromCart(id) {
            cart = cart.filter(x => x.id !== id);
            updateCartUI();
        }

        // -- SALE PROCESSING --
        function processSale() {
            if(cart.length === 0) return;
            
            let amount = cart.reduce((s, i) => s + (i.price * i.qty), 1.05);
            let orderId = Math.floor(Math.random() * 9000) + 1000;
            let time = new Date().toLocaleTimeString('ar-EG', {hour: '2-digit', minute:'2-digit'});

            salesHistory.unshift({ id: orderId, time: time, amount: amount });
            totalRevenue += amount;

            alert(`تمت العملية بنجاح! رقم الفاتورة: #${orderId}`);
            cart = [];
            updateCartUI();
        }

        // -- ADMIN DASHBOARD --
        function updateAdminDashboard() {
            document.getElementById('stat-total-sales').innerText = totalRevenue.toLocaleString() + " ج.م";
            document.getElementById('stat-bill-count').innerText = salesHistory.length;
            
            const historyBody = document.getElementById('sales-history');
            if(salesHistory.length === 0) {
                historyBody.innerHTML = `<tr><td colspan="3" class="p-10 text-center text-gray-400">لا توجد عمليات مبيعات اليوم</td></tr>`;
            } else {
                historyBody.innerHTML = salesHistory.slice(0, 5).map(s => `
                    <tr class="border-b border-gray-50 hover:bg-gray-50">
                        <td class="p-3 font-bold text-blue-600">#${s.id}</td>
                        <td class="p-3 text-center text-gray-500">${s.time}</td>
                        <td class="p-3 text-left font-black">${s.amount.toLocaleString()} ج.م</td>
                    </tr>
                `).join('');
            }
        }

        // -- CHARTING --
        const ctx = document.getElementById('salesChart').getContext('2d');
        const salesChart = new Chart(ctx, {
            type: 'line',
            data: {
                labels: ['9am', '11am', '1pm', '3pm', '5pm', '7pm', '9pm'],
                datasets: [{
                    label: 'المبيعات اللحظية',
                    data: [12, 19, 3, 5, 2, 3, 10],
                    borderColor: '#2563eb',
                    tension: 0.4,
                    fill: true,
                    backgroundColor: 'rgba(37, 99, 235, 0.1)'
                }]
            },
            options: {
                responsive: true,
                plugins: { legend: { display: false } },
                scales: { y: { display: false }, x: { grid: { display: false } } }
            }
        });

        // Initialize
        renderPOS();
    </script>
