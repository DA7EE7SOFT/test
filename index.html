-- جدول التصنيفات CREATE TABLE categories ( id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY, name TEXT NOT NULL, parent_id BIGINT NULL, created_at TIMESTAMP DEFAULT NOW() ); -- جدول المنتجات CREATE TABLE products ( id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY, barcode TEXT, name TEXT NOT NULL, price DECIMAL(10,2) NOT NULL, tax DECIMAL(5,2) DEFAULT 14, stock DECIMAL(10,2) DEFAULT 0, category_id BIGINT REFERENCES categories(id) ON DELETE SET NULL, created_at TIMESTAMP DEFAULT NOW() ); -- جدول العملاء CREATE TABLE customers ( id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY, name TEXT NOT NULL, phone TEXT, email TEXT, created_at TIMESTAMP DEFAULT NOW() ); -- جدول الفواتير CREATE TABLE invoices ( id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY, date TIMESTAMP DEFAULT NOW(), subtotal DECIMAL(10,2), tax DECIMAL(10,2), discount DECIMAL(10,2), total DECIMAL(10,2), payment_method TEXT, branch TEXT, customer_id BIGINT REFERENCES customers(id) ON DELETE SET NULL, created_at TIMESTAMP DEFAULT NOW() ); -- جدول تفاصيل الفاتورة CREATE TABLE invoice_items ( id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY, invoice_id BIGINT REFERENCES invoices(id) ON DELETE CASCADE, product_id BIGINT REFERENCES products(id) ON DELETE SET NULL, product_name TEXT, price DECIMAL(10,2), quantity DECIMAL(10,2), tax DECIMAL(5,2) ); -- أدخل عميل افتراضي INSERT INTO customers (name) VALUES ('عميل نقدي');
